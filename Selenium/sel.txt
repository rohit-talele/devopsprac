Check Your Google Chrome Version
Open Chrome → Click 3 dots (top-right) → Help → About Google Chrome

Download the Correct ChromeDriver
Go to: https://googlechromelabs.github.io/chrome-for-testing/

Find the exact version of your Chrome browser (e.g., 135.0.7049.115)

Download chromedriver-win64.zip

Extract it to a folder like:
C:\SeleniumDrivers\chromedriver.exe
---------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
Write the Python Script
Create a new file named google_search.py using Notepad or any editor.

from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.common.keys import Keys
from selenium.webdriver.chrome.service import Service
from selenium.webdriver.chrome.options import Options
import time

options = Options()
options.add_argument("--start-maximized")

chrome_driver_path = r"c:\Users\rohit\OneDrive\Desktop\devops\dump\chromedriver-win64\chromedriver.exe"

driver = webdriver.Chrome(service=Service(chrome_driver_path), options=options)

try:
    driver.get("https://www.google.com")

    search_box = driver.find_element(By.NAME, "q")
    search_box.send_keys("devops practical")

    search_box.send_keys(Keys.RETURN)

    time.sleep(2)
    print("Page Title after search:", driver.title)

finally:
    time.sleep(60)
    driver.quit()
